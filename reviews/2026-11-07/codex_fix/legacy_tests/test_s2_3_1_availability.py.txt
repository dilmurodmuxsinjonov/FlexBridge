"""Automated test suite for S2.3.1: Candidate `int4range(start_minute, end_minute, '[)')` availability rules and exceptions (`INV-01`)."""

from __future__ import annotations

from fastapi.testclient import TestClient

from app.main import app

client = TestClient(app)


def test_inv_01_half_open_availability_rules_overlap_and_adjacency() -> None:
    """Verify overlapping `[)` rules fail with 409 (`23P01`) while adjacent rules (`[540, 720)` and `[720, 900)`) succeed."""
    r1 = client.post(
        "/api/v1/candidates/cand-1/availability-rules",
        json={"weekday": 1, "start_minute": 540, "end_minute": 720},
    )
    assert r1.status_code == 201
    assert r1.json()["minute_range"] == "[540,720)"

    # Overlapping rule [660, 780) on weekday 1 must fail with 409 (`23P01`)
    overlap = client.post(
        "/api/v1/candidates/cand-1/availability-rules",
        json={"weekday": 1, "start_minute": 660, "end_minute": 780},
    )
    assert overlap.status_code == 409
    assert overlap.json()["detail"]["code"] == "OVERLAPPING_AVAILABILITY_RULE"
    assert overlap.json()["detail"]["sqlstate"] == "23P01"

    # Adjacent half-open rule [720, 900) does NOT overlap with [540, 720) and must succeed
    adjacent = client.post(
        "/api/v1/candidates/cand-1/availability-rules",
        json={"weekday": 1, "start_minute": 720, "end_minute": 900},
    )
    assert adjacent.status_code == 201
    assert adjacent.json()["minute_range"] == "[720,900)"


def test_inv_01_availability_exceptions_and_bounds() -> None:
    """Verify `availability_exceptions` rejects invalid bounds (`start >= end`) and overlapping date ranges."""
    bad_bounds = client.post(
        "/api/v1/candidates/cand-1/availability-exceptions",
        json={
            "exception_date": "2026-11-10",
            "start_minute": 800,
            "end_minute": 800,
            "kind": "UNAVAILABLE",
        },
    )
    assert bad_bounds.status_code == 422
    assert bad_bounds.json()["detail"]["code"] == "INVALID_MINUTE_RANGE"

    e1 = client.post(
        "/api/v1/candidates/cand-1/availability-exceptions",
        json={
            "exception_date": "2026-11-10",
            "start_minute": 600,
            "end_minute": 720,
            "kind": "UNAVAILABLE",
        },
    )
    assert e1.status_code == 201

    e_overlap = client.post(
        "/api/v1/candidates/cand-1/availability-exceptions",
        json={
            "exception_date": "2026-11-10",
            "start_minute": 700,
            "end_minute": 760,
            "kind": "UNAVAILABLE",
        },
    )
    assert e_overlap.status_code == 409
    assert e_overlap.json()["detail"]["sqlstate"] == "23P01"
