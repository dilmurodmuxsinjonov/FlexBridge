"""Automated test suite for S2.1.1: Backend foundation, config, health APIs, and 1.4 DB schema invariants."""

from __future__ import annotations

from fastapi.testclient import TestClient

from app.core.config import FORBIDDEN_PDF_TOKENS, settings
from app.db.schema import (

def test_api_x05_health_live() -> None:
    """Verify API-X05 (`GET /api/v1/health/live`) returns 200 and alive status."""
    response = client.get("/api/v1/health/live")
    assert response.status_code == 200
    payload = response.json()
    assert payload["status"] == "alive"
    assert payload["contract_id"] == "API-X05"
    assert payload["version"] == "1.4.0"


def test_api_x06_health_ready() -> None:
    """Verify API-X06 (`GET /api/v1/health/ready`) validates 94/935/214 counts and 17/17 PG checks."""

    payload = response.json()

    assert payload["schema_counts"]["reference_errors"] == 0
    assert payload["postgres_verification"]["all_checks_passed"] is True
    assert payload["postgres_verification"]["passed_checks"] == 17

def test_db_083_retention_policies_v1() -> None:
    """Verify all 8 legal retention policies (`DB-083`) from 1.4 PDF section 6 are present."""
    policies = get_retention_policies_v1()
    assert len(policies) == 8
    codes = tuple(p["data_class"] for p in policies)
    assert codes == RETENTION_POLICY_V1_CODES

    by_class = {p["data_class"]: p for p in policies}
    assert by_class["BILLING_AND_FISCAL_LEDGER"]["active_days"] == 1825
    assert by_class["INTERNSHIP_DIARY_ACADEMIC"]["active_days"] == 1095
    assert by_class["OTP_CHALLENGE_DIGEST"]["active_days"] == 1


def test_postgres_1_4_invariant_report_completeness() -> None:
    """Verify all 17 PostgreSQL invariant checks passed in `v1_4_schema_verification.json`."""
    report = load_verification_report()
    assert report["all_checks_passed"] is True
    assert report["passed_checks"] == 17
    assert report["total_checks"] == 17
    intro = report["database_introspection"]
    assert intro["base_tables"] == 94
    assert intro["non_generated_columns"] == 935
    assert intro["generated_stored_columns"] == 4
    assert intro["retention_policy_v1_rows"] == 8

    summary_resp = client.get("/api/v1/meta/schema-summary")
    assert summary_resp.status_code == 200
    assert len(summary_resp.json()["retention_policies_v1"]) == 8
